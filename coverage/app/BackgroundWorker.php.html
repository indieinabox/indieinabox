746    /**
747     * Resolves a relative URL against a base URL.
748     */
749    private function resolveUrl(string $base, string $rel): string
750    {
751        if (parse_url($rel, PHP_URL_SCHEME) != '') return $rel;
752        if ($rel[0] == '#' || $rel[0] == '?') return $base . $rel;
753        extract(parse_url($base));
754        /** @var string $scheme */
755        /** @var string $host */
756        /** @var string $path */
757        if (!isset($path)) $path = '/';
758        $path = preg_replace('#/[^/]*$#', '', $path);
759        if ($rel[0] == '/') $path = '';
760        $abs = "$host$path/$rel";
761        $re = ['#(/\.?/)#', '#/(?!\.\.)[^/]+/\.\./#'];
762        for ($n = 1; $n > 0; $abs = preg_replace($re, '/', $abs, -1, $n)) {}
763        return $scheme . '://' . $abs;
764    }
